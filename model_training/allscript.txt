def pso_optimize(val_probs, y_val, swarm_size=30, iterations=100, seed=None):
    rng = np.random.RandomState(seed)
    n = val_probs.shape[0]
    w_start, w_end = 0.7, 0.4
    c1, c2 = 1.5, 1.5
    v_max = 0.2

    pos = rng.rand(swarm_size, n)
    pos = pos / pos.sum(axis=1, keepdims=True)
    vel = rng.uniform(-v_max, v_max, (swarm_size, n))

    def fitness(w):
        return roc_auc_score(y_val, np.average(val_probs, axis=0, weights=w))

    fit = np.array([fitness(p) for p in pos])
    pbest_pos = pos.copy()
    pbest_fit = fit.copy()
    gbest_idx = np.argmax(fit)
    gbest_pos = pos[gbest_idx].copy()
    gbest_fit = fit[gbest_idx]
    history = [gbest_fit]

    for t in range(iterations):
        w = w_start - (w_start - w_end) * (t / iterations)
        for i in range(swarm_size):
            r1, r2 = rng.rand(n), rng.rand(n)
            vel[i] = w*vel[i] + c1*r1*(pbest_pos[i]-pos[i]) + c2*r2*(gbest_pos-pos[i])
            vel[i] = np.clip(vel[i], -v_max, v_max)
            pos[i] = np.clip(pos[i] + vel[i], 0, 1)
            s = pos[i].sum()
            pos[i] = pos[i]/s if s > 0 else np.ones(n)/n
            fit[i] = fitness(pos[i])
            if fit[i] > pbest_fit[i]:
                pbest_fit[i] = fit[i]; pbest_pos[i] = pos[i].copy()
            if fit[i] > gbest_fit:
                gbest_fit = fit[i]; gbest_pos = pos[i].copy()
        history.append(gbest_fit)
    return gbest_pos, gbest_fit, history

"""
PSO-Optimized Model Selection for Stacking Ensemble
====================================================
Uses Particle Swarm Optimization to find optimal ensemble weights,
then selects the top 3 models by weight and trains a Gradient Boosting
meta-learner for the final stacking ensemble.

Pipeline:
  1. Load/generate base model predictions on val & test sets
  2. PSO: find optimal continuous weights for all 5 models (maximize val AUC)
  3. Extract top 3 models by PSO weight
  4. Train Gradient Boosting meta-learner on top-3 subset
  5. Compare: PSO-weighted ensemble, PSO-3 stacking, all-5 stacking, voting

PSO Parameters:
  - Swarm size: 30 particles
  - Iterations: 100
  - Inertia: 0.7 → 0.4 (linearly decaying)
  - Cognitive/Social coefficients: c1=1.5, c2=1.5
  - Fitness: AUC on validation set
  - Bounds: [0, 1] per weight, normalized to sum=1
"""

# =========================================================
# PSO IMPLEMENTATION (manual, no external dependencies)
# =========================================================
def pso_optimize_weights(val_probs, y_val, swarm_size, iterations, seed=None):
    """
    Particle Swarm Optimization for ensemble weight optimization.

    Each particle is a 5D weight vector. Fitness = AUC on validation set.
    Weights are normalized to sum=1 after each update.

    Returns: (best_weights, best_fitness, history)
    """
    rng = np.random.RandomState(seed)
    n_models = val_probs.shape[0]

    # PSO hyperparameters
    w_start, w_end = 0.7, 0.4        # inertia (linear decay)
    c1, c2 = 1.5, 1.5               # cognitive, social coefficients
    v_max = 0.2                       # velocity clamping

    # Initialize swarm
    positions = rng.rand(swarm_size, n_models)
    positions = positions / positions.sum(axis=1, keepdims=True)  # normalize

    velocities = rng.uniform(-v_max, v_max, (swarm_size, n_models))

    # Evaluate initial fitness
    def compute_fitness(weights):
        """Weighted ensemble AUC on validation set."""
        ensemble_probs = np.average(val_probs, axis=0, weights=weights)
        return roc_auc_score(y_val, ensemble_probs)

    fitness = np.array([compute_fitness(p) for p in positions])

    pbest_positions = positions.copy()
    pbest_fitness = fitness.copy()

    gbest_idx = np.argmax(fitness)
    gbest_position = positions[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]

    history = [gbest_fitness]

    # Main PSO loop
    for iter_idx in range(iterations):
        w = w_start - (w_start - w_end) * (iter_idx / iterations)

        for i in range(swarm_size):
            r1, r2 = rng.rand(n_models), rng.rand(n_models)

            # Velocity update
            velocities[i] = (
                w * velocities[i]
                + c1 * r1 * (pbest_positions[i] - positions[i])
                + c2 * r2 * (gbest_position - positions[i])
            )

            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)

            # Position update
            positions[i] = positions[i] + velocities[i]

            # Clamp and normalize
            positions[i] = np.clip(positions[i], 0, 1)
            pos_sum = positions[i].sum()
            if pos_sum > 0:
                positions[i] = positions[i] / pos_sum
            else:
                positions[i] = np.ones(n_models) / n_models

            # Evaluate
            fitness[i] = compute_fitness(positions[i])

            # Update personal best
            if fitness[i] > pbest_fitness[i]:
                pbest_fitness[i] = fitness[i]
                pbest_positions[i] = positions[i].copy()

            # Update global best
            if fitness[i] > gbest_fitness:
                gbest_fitness = fitness[i]
                gbest_position = positions[i].copy()

        history.append(gbest_fitness)

    return gbest_position, gbest_fitness, history

"""
Stacking Ensemble for BreastVisionAI
=====================================
Implements prediction-level stacking with Gradient Boosting as the
single meta-learner. The meta-learner is trained on OOF (out-of-fold)
predictions from the five CNN base learners on the validation set,
then evaluated on the held-out test set.

CRITICAL: Uses validation set for OOF (out-of-fold) meta-learner training,
          then evaluates on held-out test set to prevent data leakage.

Expected directory layout (run from project root):
  BreastVisionAI/
    models/          -- all 5 .keras models
    datasets_split/  -- train/val/test splits
"""
import os
import sys
import warnings
import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf

from tensorflow.keras.preprocessing.image import ImageDataGenerator
from tensorflow.keras.applications.efficientnet import preprocess_input as eff_pre
from tensorflow.keras.applications.densenet import preprocess_input as den_pre
from tensorflow.keras.applications.resnet import preprocess_input as res_pre
from tensorflow.keras.applications.vgg16 import preprocess_input as vgg_pre
from tensorflow.keras.applications.xception import preprocess_input as xcep_pre

from sklearn.ensemble import GradientBoostingClassifier
from sklearn.metrics import (
    accuracy_score, precision_score, recall_score,
    f1_score, roc_auc_score, roc_curve
)

warnings.filterwarnings('ignore')

# =========================================================
# PATHS
# =========================================================
base_dir = 'datasets_split'
val_dir = os.path.join(base_dir, 'val')
test_dir = os.path.join(base_dir, 'test')

output_dir = 'Stacking_Result'
os.makedirs(output_dir, exist_ok=True)

# =========================================================

# =========================================================
img_size = (224, 224)
batch_size = 32

# =========================================================

# =========================================================
models_info = {
    "EfficientNet": {"path": "models/efficientnet_final_model.keras", "preprocess": eff_pre},
    "DenseNet":    {"path": "models/densenet_final_model.keras",    "preprocess": den_pre},
    "ResNet":      {"path": "models/resnet_final_model.keras",      "preprocess": res_pre},
    "VGG16":       {"path": "models/vgg16_final_model.keras",       "preprocess": vgg_pre},

    generator = datagen.flow_from_directory(
        val_dir,
        target_size=img_size,
        batch_size=batch_size,
        class_mode='binary',
        shuffle=False
    )

    if y_val is None:
        y_val = generator.classes

    print(f"  Predicting {name} on {len(y_val)} validation samples...")
    probs = model.predict(generator, verbose=0).flatten()
    val_probs[name] = probs
    del model  # free memory

print(f"\n  Validation set: {len(y_val)} samples")
print(f"  Class distribution: benign={np.sum(y_val == 0)}, malignant={np.sum(y_val == 1)}")

# GENERATE PREDICTIONS ON TEST SET

print("\n" + "=" * 60)
print("STEP 2: Generating predictions on TEST set")

        "precision": precision_score(y_true, y_pred),
        "recall": recall_score(y_true, y_pred),
        "specificity": recall_score(y_true, y_pred, pos_label=0),
        "f1": f1_score(y_true, y_pred),

    }

}
